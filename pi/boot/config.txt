# littlelx - Raspberry Pi 3B touchscreen front panel
# The Pi never writes to this card. Edit this file from your computer if needed.

start_file=start_cd.elf
fixup_file=fixup_cd.dat

# Which copy of the OS to boot: a/ or b/. Updates over USB write the other
# copy and then change this one letter. If an update ever leaves the panel
# dead, change it back by hand.
os_prefix=a/
kernel=zImage
initramfs littlelx.cpio.gz followkernel
gpu_mem=16
boot_delay=0
disable_splash=1
disable_overscan=1
camera_auto_detect=0
display_auto_detect=0

# ---- Display: 3.5" 480x320 ILI9486 + XPT2046 touch on the GPIO header ----
# littlelx35 = the generic "3.5inch RPi Display" boards (goodtft LCD-show
# "LCD35-show" / tft35a / MPI3501 and clones). Genuine Waveshare 3.5" (A)
# boards use piscreen instead: swap the # between the two lines below.
# rotate=0 or 180 = portrait, 90 or 270 = landscape (re-run the touch
# calibration after changing it). Want it snappier? try speed=24000000 or
# 32000000; if the picture breaks up, go back down.
dtparam=spi=on
dtoverlay=littlelx35,speed=16000000,rotate=0
#dtoverlay=piscreen,speed=16000000,rotate=0

# ---- UART to the Mega: GPIO14 = Pi TX, GPIO15 = Pi RX, 500000 baud ----
# Bluetooth gives the good UART back to the header; wifi is off to save power.
enable_uart=1
dtoverlay=disable-bt
dtoverlay=disable-wifi

# ---- Performance ----
# Fixed clocks so the SPI clock never changes mid-frame, plus a mild overclock.
# NOTE: the firmware still drops to 600MHz whenever it sees under-voltage, which
# is what a bus-powered Pi 3 does. A solid 5V 2.5A supply is the real speed-up.
# If the Pi ever locks up or reboots randomly, delete these five lines first.
force_turbo=1
arm_freq=1300
core_freq=450
sdram_freq=450
over_voltage=2

# Green LED: double-blinking heartbeat = Linux is running.
# (No blink at all = it isn't booting: try removing the overclock lines.)
dtparam=act_led_trigger=heartbeat
dtparam=audio=off
